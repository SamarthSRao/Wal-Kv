v — Simple Demo UI
Talks to /api/put, /api/get, /api/delete
Use the controls to PUT/GET/DELETE keys. Writes are appended to the WAL before being applied.
Result: —

Interactive

Tip: PUT> and DELETE use the WAL for durability. Try writing a few keys, then stop the server and restart it to observe recovery (server must be run separately).
HTTP usage:
  • PUT /
    • api/put?key=KEY&value=VALUE (method: PUT)
  • GET /api/get?key=
    • KEY (method: GET)
  • DELETE /api/delete
    • ?key=KEY (method: DELETE)

const value = document.getElementById('value').value; if (!key) { setResult(false, 'missing key'); return; } appendLog(`PUT ${key}=${value} ...`); try { // Server expects PUT with query params const url = `/api/put?key=${encodeURIComponent(key)}&value=${encodeURIComponent(value)}`; const res = await fetch(url, { method: 'PUT' }); const json = await res.json(); if (json.ok) { setResult(true, ''); appendLog(`PUT OK: ${key}`); } else { setResult(false, json.error || 'unknown error'); if (json.ok) { setResult(true, json.value || ''); appendLog(`GET OK: ${key} => ${json.value}`); } else { setResult(false, json.error || 'not found'); appendLog(`GET: ${json.error || 'not found'}`); } } catch (err) { setResult(false, String(err));
